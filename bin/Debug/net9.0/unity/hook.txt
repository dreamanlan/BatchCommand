{
	onenter{
		// argv layout (DBGSCP_HOOK with bool ret):
		//   argv(0)  = &hMismatch (int)
		//   argv(1)  = &hCount
		//   argv(2)  = &hMipLevel
		//   argv(3)  = &hFace
		//   argv(4)  = &hDepthSlice

		$count:int      = ptrget(argv(1),4);
		$mipLevel:int   = ptrget(argv(2),4);
		$face:int       = ptrget(argv(3),4);

		$pRtW:int       = ptrget(argv(11),8);
		$pRtH:int       = ptrget(argv(12),8);

		if($mismatch!=0){
			$str:string = format("SetRenderTargets_MainThread mismatch time:%lld tid:%d count:%d mip:%d face:%d depthSlice:%d ds(tex:%d fmt:%d %dx%d)\n",time(),GetTID(),$count,$mipLevel,$face,$depthSlice,$dsTexId,$dsFormat,$dsW,$dsH);

			};

				TriggerFrameCapture();
				@g_capture_frames = @g_capture_frames + 1;
			};
			//assert(0);
		};
		return 0;
	};

//------ D3D12 shader bytecode + PSO dump (AMD shader hash collision analysis) ------
// Shared switch. 1 = enable both dumps. Reload this script (hot reload) to apply.
// Outputs:
//   engine file (persistent data path): shader_bytecode_dump.bin
//     self-describing records: [UInt64 bytecodePtr][UInt64 length][content bytes]
//   dbgscp_log lines (written by these scripts):
//     SHADER_BC ptr:<hex> stage:<0|1> len:<n> name:<shader name>
//     PSO_CREATED hash:<16hex> vs:<ptr>/<len> ps:<ptr>/<len> gs:<ptr>/<len> hs:<ptr>/<len> ds:<ptr>/<len> rt:<ptr>/<len>
@g_dump_shader_bytecode : int = 1;

hook("D3D12::ShaderBytecode")
{
	onenter{
		if(@g_dump_shader_bytecode == 0){
			return 0;
		};
		$stage:int = ptrget(argv(1),4);
		$ptr:int = ptrget(argv(2),8);
		$len:int = ptrget(argv(3),8);
		$name:string = ptrget(argv(4),8);
		$str:string = format("SHADER_BC ptr:%llx stage:%d len:%lld name:%s\n",$ptr,$stage,$len,$name);
		printf("%s",$str);
		WriteLog($str);
		ptrset(argv(0),1,1); // tell C++ to append this bytecode to the bin file
	};
};
